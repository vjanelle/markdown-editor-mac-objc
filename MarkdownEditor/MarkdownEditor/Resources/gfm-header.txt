<script src="mermaid.min.js"></script>
<script>
  (function() {
    function mermaidSourceNode(root) {
      return root.querySelector('pre > code.language-mermaid, pre > code.mermaid, code.language-mermaid, code.mermaid, pre.mermaid');
    }

    function promoteMermaidBlocks() {
      var nodeList = document.querySelectorAll('pre > code.language-mermaid, pre > code.mermaid, pre.mermaid');
      nodeList.forEach(function(node) {
        var pre = node.tagName === 'PRE' ? node : node.parentElement;
        if (!pre || pre.dataset.mermaidProcessed === 'true') {
          return;
        }
        var container = document.createElement('div');
        container.className = 'mermaid';
        container.textContent = node.textContent;
        pre.dataset.mermaidProcessed = 'true';
        pre.replaceWith(container);
      });
    }

    function renderMermaid() {
      if (!window.mermaid || !mermaidSourceNode(document)) {
        return;
      }
      promoteMermaidBlocks();
      window.mermaid.initialize({
        startOnLoad: false,
        securityLevel: 'strict',
        theme: window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'default'
      });
      window.mermaid.run({ querySelector: '.mermaid' }).catch(function(error) {
        console.error('Failed to render Mermaid diagram.', error);
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', renderMermaid);
    } else {
      renderMermaid();
    }
  })();
</script>
